서식 지정자 익히기

출력 
#include <stdio.h>
int main() {
    int a = 10;
    double b = 3.14159;
    char c = 'A';
    char s[] = "Info";
    printf("%d|%5d|%-5d|\n", a, a, a);
    printf("%f %.2f\n", b, b);
    printf("%c %d\n", c, c);
    printf("%s\n", s);
    return 0;
}

    b = ++a;

포인터로 값 바꾸기 swap

포인터 
#include <stdio.h>
void swap(int *x, int *y) {
    int t = *x; *x = *y; *y = t;
}
void wrong(int x, int y) {
    int t = x; x = y; y = t;
}
int main() {
    int a = 1, b = 2;
    wrong(a, b);
    printf("%d %d\n", a, b);
    swap(&a, &b);
    printf("%d %d\n", a, b);
    return 0;
}
1 2
2 1

값 전달은 원본이 안 바뀌고, 주소 전달은 바뀐다. 이 대비가 시험 포인트.
